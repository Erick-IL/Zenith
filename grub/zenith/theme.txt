# The GRUB theme "Zenith" is the same language as the dynamic island:
# Fav theme wallpaper, highlight capsule on the selected item, nothing extra.
#
# Font names are written exactly as they are inside .pf2, and not as
# they were called during assembly: grub-mkfont adds a style to the family, and from
# "Zenith Regular" becomes "Zenith Regular Regular". Didn't match at least
# letter - GRUB silently does not find the font and draws a simple menu instead of the theme, without
# single error message. Check the name inside the file:
#   strings zenith18.pf2 | head -3

desktop-image: "background.png"
desktop-color: "#050506"

title-text: ""

# Console box (C key) and bootloader service messages. Stands below and
# low: in it GRUB prints “Loading Linux...” before transferring
# kernel control, and before it looked like a black window in the middle of the screen -
# the box occupied the center and was almost opaque.
terminal-font: "Zenith Regular Regular 18"
terminal-box: "menu_*.png"
terminal-left: "12%"
terminal-top: "72%"
terminal-width: "76%"
terminal-height: "24%"

message-font: "Zenith Regular Regular 18"
message-color: "#f2f2f2"
message-bg-color: "#050506"

    # GRUB draws highlight AROUND the line: padding is added to its height
    # frames equal to the corner image (12 px top and bottom). Therefore the gap

    item_height = 40
    item_spacing = 28

}
